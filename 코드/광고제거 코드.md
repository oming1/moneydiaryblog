# =================================================================
# 모든 마크다운 파일에서 애드센스 광고 코드 완전 제거 스크립트
# =================================================================

$contentPath = "D:\GitHub\moneydiaryblog\content"

Write-Host "========================================" -ForegroundColor Cyan
Write-Host "   애드센스 광고 코드 완전 제거" -ForegroundColor Cyan
Write-Host "========================================" -ForegroundColor Cyan
Write-Host "대상 경로: $contentPath"
Write-Host ""
Write-Host "🚨 경고: 이 작업은 파일 내용을 직접 수정합니다. 백업 필수! 🚨" -ForegroundColor Red
Write-Host ""

$totalFilesProcessed = 0
$totalAdsRemoved = 0

Get-ChildItem -Path $contentPath -Filter "*.md" -Recurse -File | ForEach-Object {
    $filePath = $_.FullName
    $fileName = $_.Name
    
    $content = Get-Content -Path $filePath -Raw -Encoding UTF8
    
    # 광고 코드가 있는지 확인
    if ($content -match '\{\{<\s*adsense-infeed\s*>\}\}') {
        Write-Host "처리 중: $fileName" -ForegroundColor Cyan
        
        # 광고 코드 개수 세기
        $adsCount = ([regex]::Matches($content, '\{\{<\s*adsense-infeed\s*>\}\}')).Count
        
        # 패턴 1: 빈 줄 + 광고 코드 + 빈 줄
        $newContent = $content -replace '\n\s*\{\{<\s*adsense-infeed\s*>\}\}\s*\n', "`n"
        
        # 패턴 2: 광고 코드만 있는 경우
        $newContent = $newContent -replace '\{\{<\s*adsense-infeed\s*>\}\}', ''
        
        # 패턴 3: 광고 코드 앞에만 빈 줄
        $newContent = $newContent -replace '\n\s*\{\{<\s*adsense-infeed\s*>\}\}', ''
        
        # 패턴 4: 광고 코드 뒤에만 빈 줄
        $newContent = $newContent -replace '\{\{<\s*adsense-infeed\s*>\}\}\s*\n', ''
        
        # 연속된 빈 줄 정리 (3개 이상의 빈 줄을 2개로)
        $newContent = $newContent -replace '\n\n\n+', "`n`n"
        
        # 파일 저장
        Set-Content -Path $filePath -Value $newContent -Encoding UTF8 -NoNewline
        
        $totalFilesProcessed++
        $totalAdsRemoved += $adsCount
        Write-Host "  ✓ 완료: $adsCount 개 광고 제거" -ForegroundColor Green
    }
}

Write-Host ""
Write-Host "========================================" -ForegroundColor Magenta
Write-Host "   광고 제거 완료!" -ForegroundColor Magenta
Write-Host "   총 $totalFilesProcessed 개의 파일 처리" -ForegroundColor Magenta
Write-Host "   총 $totalAdsRemoved 개의 광고 제거" -ForegroundColor Magenta
Write-Host "========================================" -ForegroundColor Magenta
Write-Host ""
Write-Host "다음 단계:" -ForegroundColor Cyan
Write-Host "1. hugo server -D 로 로컬 테스트" -ForegroundColor White
Write-Host "2. git add ." -ForegroundColor White
Write-Host "3. git commit -m 'Remove all adsense-infeed shortcodes'" -ForegroundColor White
Write-Host "4. git push origin main" -ForegroundColor White
