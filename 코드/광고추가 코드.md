# =================================================================
# Hugo 마크다운 파일에 애드센스 인피드 광고 자동 삽입 스크립트 (여백 포함)
# =================================================================
# 
# 기능: 각 .md 파일의 문단 사이(빈 줄)에 {{< adsense-infeed >}}를 3~5개 랜덤 삽입
#       광고 코드 위아래에 빈 줄 추가하여 깔끔하게 삽입
#
# =================================================================

# --- 사용자 설정 ---
$contentPath = "D:\GitHub\moneydiaryblog\content"
# 위 경로를 당신의 실제 content 폴더 경로로 수정하세요.

$minAds = 2  # 파일당 최소 광고 개수
$maxAds = 4  # 파일당 최대 광고 개수

Write-Host "========================================" -ForegroundColor Cyan
Write-Host "   애드센스 인피드 광고 자동 삽입 스크립트" -ForegroundColor Cyan
Write-Host "========================================" -ForegroundColor Cyan
Write-Host "대상 경로: $contentPath"
Write-Host "파일당 광고 개수: $minAds ~ $maxAds 개"
Write-Host ""
Write-Host "🚨 경고: 이 작업은 파일 내용을 직접 수정합니다. 백업 필수! 🚨" -ForegroundColor Red
Write-Host ""

$totalFilesProcessed = 0
$totalAdsInserted = 0

# content 폴더 내 모든 .md 파일 재귀적으로 탐색
Get-ChildItem -Path $contentPath -Filter "*.md" -Recurse -File | ForEach-Object {
    $filePath = $_.FullName
    $fileName = $_.Name
    
    # _index.md 파일은 건너뛰기 (섹션 인덱스 파일)
    if ($fileName -eq "_index.md") {
        return
    }
    
    $content = Get-Content -Path $filePath -Raw -Encoding UTF8
    
    # Front Matter 분리 (--- 사이의 YAML 부분)
    $frontMatterMatch = [regex]::Match($content, '(?s)^---\s*\n(.*?)\n---\s*\n([\s\S]*)$')
    
    if (-not $frontMatterMatch.Success) {
        Write-Host "⚠ Front Matter를 찾을 수 없음: $fileName (건너뜀)" -ForegroundColor Yellow
        return
    }
    
    $frontMatter = $frontMatterMatch.Groups[1].Value
    $bodyContent = $frontMatterMatch.Groups[2].Value
    
    # 본문이 너무 짧으면 건너뛰기 (최소 500자 이상)
    if ($bodyContent.Length -lt 500) {
        Write-Host "⚠ 본문이 너무 짧음: $fileName (건너뜀)" -ForegroundColor Yellow
        return
    }
    
    # 이미 광고가 삽입되어 있으면 건너뛰기
    if ($bodyContent -match '\{\{<\s*adsense-infeed\s*>\}\}') {
        Write-Host "⚠ 이미 광고가 있음: $fileName (건너뜀)" -ForegroundColor Yellow
        return
    }
    
    Write-Host "처리 중: $fileName" -ForegroundColor Cyan
    
    # 본문을 줄 단위로 분리
    $lines = $bodyContent -split "`n"
    
    # 안전한 삽입 위치 찾기 (빈 줄 다음 줄)
    $safePositions = @()
    $inCodeBlock = $false
    
    for ($i = 0; $i -lt $lines.Count; $i++) {
        $line = $lines[$i]
        
        # 코드 블록 시작/종료 감지 (```)
        if ($line -match '^```') {
            $inCodeBlock = -not $inCodeBlock
        }
        
        # 안전한 위치: 빈 줄이고, 코드 블록 밖이고, 다음 줄도 빈 줄이 아닌 경우
        if (-not $inCodeBlock -and $line.Trim() -eq "" -and $i -lt ($lines.Count - 1) -and $lines[$i + 1].Trim() -ne "") {
            # 최소 10줄 이후부터 삽입 가능 (Front Matter 직후 방지)
            # 그리고 Front Matter 종료(---) 바로 다음 줄은 피하기
            if ($i -gt 10 -and $lines[$i - 1].Trim() -ne "---") {
                $safePositions += $i
            }
        }
    }
    
    # 안전한 위치가 충분하지 않으면 건너뛰기
    if ($safePositions.Count -lt $minAds) {
        Write-Host "  ⚠ 안전한 삽입 위치 부족 ($($safePositions.Count)개): $fileName (건너뜀)" -ForegroundColor Yellow
        return
    }
    
    # 랜덤하게 광고 개수 결정
    $numAds = Get-Random -Minimum $minAds -Maximum ($maxAds + 1)
    
    # 삽입할 위치를 랜덤하게 선택 (중복 없이)
    $selectedPositions = $safePositions | Get-Random -Count ([Math]::Min($numAds, $safePositions.Count)) | Sort-Object
    
    # 뒤에서부터 삽입 (인덱스가 밀리지 않도록)
    $selectedPositions = $selectedPositions | Sort-Object -Descending
    
    foreach ($pos in $selectedPositions) {
        # 빈 줄 다음에 광고 코드 삽입 (위아래 빈 줄 포함)
        # 형식: (빈 줄) + (광고 코드) + (빈 줄)
        $lines[$pos] = "`n{{< adsense-infeed >}}`n"
        $totalAdsInserted++
    }
    
    # 본문 재조립
    $newBodyContent = $lines -join "`n"
    
    # Front Matter + 본문 결합
    $newContent = @"
---
$frontMatter
---
$newBodyContent
"@
    
    # 파일 저장
    Set-Content -Path $filePath -Value $newContent -Encoding UTF8 -NoNewline
    
    $totalFilesProcessed++
    Write-Host "  ✓ 완료: $($selectedPositions.Count)개 광고 삽입" -ForegroundColor Green
}

Write-Host ""
Write-Host "========================================" -ForegroundColor Magenta
Write-Host "   광고 삽입 완료!" -ForegroundColor Magenta
Write-Host "   총 $totalFilesProcessed 개의 파일 처리" -ForegroundColor Magenta
Write-Host "   총 $totalAdsInserted 개의 광고 삽입" -ForegroundColor Magenta
Write-Host "========================================" -ForegroundColor Magenta
Write-Host ""
Write-Host "이제 다음 단계를 진행하세요:" -ForegroundColor Cyan
Write-Host "1. hugo server -D 로 로컬 테스트" -ForegroundColor White
Write-Host "2. 문제 없으면 git add ." -ForegroundColor White
Write-Host "3. git commit -m 'Add AdSense infeed ads to content'" -ForegroundColor White
Write-Host "4. git push origin main" -ForegroundColor White
