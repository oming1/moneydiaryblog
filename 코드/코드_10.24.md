# 5. [[image.png]] 형태의 위키링크를 ![Alt Text](/images/image.png) 형태로 변환
# 6. ![](Pasted image.png) 형태를 ![](/images/Pasted image.png) 형태로 변환 (간단한 경로 수정)

# 1. 프로젝트의 content 폴더 경로 (반드시 실제 경로로 수정)
$contentPath = "D:\GitHub\moneydiaryblog\content"

# 2. 프로젝트의 static 폴더 경로 (반드시 실제 경로로 수정)
$staticPath = "D:\GitHub\moneydiaryblog\static"

# 3. static 폴더 내 이미지 폴더명 (일반적으로 "images")
#    예: static/images/ 에 이미지가 있다면 "images"로 설정
$imagesStaticSubPath = "images"

# 4. 웹사이트 상에서 이미지가 접근될 기본 경로
$webImagePath = "/images"

# 5. 처리할 섹션과 해당 permalinks 매핑
$sectionPermalinks = @{
    "산업-테마공부" = "/industry-study/"
    "기업공부" = "/company-analysis/"
    "Daily시황정리" = "/daily-summary/"
    "earning" = "/earnings/"
    "투자아이디어" = "/investment-ideas/"
}

# 6. 스크립트가 처리할 최상위 섹션 폴더 이름 목록
$sections = @("Daily시황정리", "산업-테마공부", "기업공부", "earning", "투자아이디어")

# =================================================================

Write-Host "   Hugo 마크다운 파일 완전 자동화 스크립트" -ForegroundColor Cyan
Write-Host "   (올인원 최종 버전 - 사이드바 설정 포함)" -ForegroundColor Cyan
Write-Host "========================================" -ForegroundColor Cyan
Write-Host "대상 경로: $contentPath"
Write-Host "정적 파일 경로: $staticPath"
Write-Host "이미지 웹 경로: $webImagePath"
Write-Host "처리할 섹션: $($sections -join ', ')"

# URL 안전성을 위한 함수 정의
function Get-SafeSlug {
    param([string]$text)
    
    $slug = $text.ToLower()
    $slug = $slug -replace '[^a-z0-9가-힣\s\-_]', ''
    $slug = $slug -replace '\s+', '-'
    $slug = $slug -replace '-+', '-'
    $slug = $slug.Trim('-')
    
    return $slug
}

# =================================================================
# 1단계: 누락된 _index.md 파일들 자동 생성
# =================================================================
Write-Host "=== 1단계: 누락된 _index.md 파일 생성 ===" -ForegroundColor Magenta

foreach ($section in $sections) {
    $sectionPath = Join-Path $contentPath $section
    
    if (Test-Path $sectionPath) {
        Write-Host "섹션 스캔 중: $section" -ForegroundColor Green
        
        $allFolders = @($sectionPath) + (Get-ChildItem -Path $sectionPath -Directory -Recurse | Select-Object -ExpandProperty FullName)
        
        foreach ($folderPath in $allFolders) {
            $indexPath = Join-Path $folderPath "_index.md"
            
            if (-not (Test-Path $indexPath)) {
                $folderName = Split-Path $folderPath -Leaf
                $relativePath = $folderPath.Replace($contentPath, "").TrimStart('\')
                
                $indexContent = @"
---
title: $folderName
---

# $folderName

이 섹션의 내용입니다.
"@
                
                Set-Content -Path $indexPath -Value $indexContent -Encoding UTF8
                Write-Host "  ✓ 생성됨: /$relativePath/_index.md" -ForegroundColor Green
            }
        }
    }
}

Write-Host ""

# =================================================================
# 2단계: Permalinks를 고려한 URL 매핑 생성 (이미지 파일 포함)
# =================================================================
Write-Host "=== 2단계: Permalinks 기반 URL 매핑 생성 (이미지 포함) ===" -ForegroundColor Magenta

$linkMap = @{}

# 2-1: .md 파일 매핑
Write-Host "Markdown 파일 매핑 중..." -ForegroundColor Green
Get-ChildItem -Path $contentPath -Filter "*.md" -Recurse -File | ForEach-Object {
    $filePath = $_.FullName
    $fileName = $_.BaseName
    $relativeContentPath = $filePath.Replace($contentPath, "").Replace("\", "/").TrimStart('/')
    
    # 섹션 확인
    $section = ""
    foreach ($sectionName in $sections) {
        if ($relativeContentPath.StartsWith($sectionName + "/") -or $relativeContentPath -eq $sectionName) {
            $section = $sectionName
            break
        }
    }
    
    if (-not $section) {
        return
    }
    
    # Permalinks 기반 URL 생성
    $basePermalink = $sectionPermalinks[$section]
    $hugoUrl = ""
    
    if ($fileName -eq "_index") {
        $folderPath = Split-Path $relativeContentPath -Parent
        if ($folderPath -eq $section) {
            $hugoUrl = $basePermalink
        } else {
            $subPath = $folderPath.Replace($section, "").Trim('/')
            if ($subPath) {
                $subSlugParts = $subPath.Split('/') | ForEach-Object { Get-SafeSlug $_ }
                $hugoUrl = $basePermalink + ($subSlugParts -join '/') + "/"
            } else {
                $hugoUrl = $basePermalink
            }
        }
    } else {
        $fileSlug = Get-SafeSlug $fileName
        
        $fileFolder = Split-Path $relativeContentPath -Parent
        $pathFromSection = $fileFolder.Replace($section, "").Trim('/')
        if ($pathFromSection) {
            $pathFromSectionSlugParts = $pathFromSection.Split('/') | ForEach-Object { Get-SafeSlug $_ }
            $hugoUrl = $basePermalink + ($pathFromSectionSlugParts -join '/') + "/" + $fileSlug + "/"
        } else {
            $hugoUrl = $basePermalink + $fileSlug + "/"
        }
    }

    # 링크 타겟 생성
    $linkTargets = @()
    $linkTargets += $fileName
    $linkTargets += $fileName -replace '[-_]', ' '
    
    if ($fileName -eq "_index") {
        $folderName = (Split-Path $filePath -Parent).Split('\')[-1]
        if ($folderName) {
            $linkTargets += $folderName
            $linkTargets += $folderName -replace '[-_]', ' '
        }
    }
    
    $pathAsLinkTarget = $relativeContentPath -replace "\.md$", ""
    if ($pathAsLinkTarget -ne "/" -and $pathAsLinkTarget -ne "") {
        $linkTargets += $pathAsLinkTarget
        $linkTargets += $pathAsLinkTarget -replace '[-_]', ' '
    }
    
    foreach ($target in $linkTargets | Select-Object -Unique) {
        if ($target -ne "") {
            if (-not $linkMap.ContainsKey($target) -or $linkMap[$target].Length -lt $hugoUrl.Length) {
                $linkMap[$target] = $hugoUrl
            }
        }
    }
}

# 2-2: 이미지 파일 매핑 추가
$imagesFullPath = Join-Path $staticPath $imagesStaticSubPath

if (Test-Path $imagesFullPath) {
    Write-Host "이미지 파일 매핑 중..." -ForegroundColor Green
    
    $imageExtensions = @("*.png", "*.jpg", "*.jpeg", "*.gif", "*.webp", "*.svg", "*.bmp")
    
    foreach ($extension in $imageExtensions) {
        Get-ChildItem -Path $imagesFullPath -Filter $extension -Recurse -File | ForEach-Object {
            $imageFile = $_
            $imageName = $imageFile.Name
            
            $imageUrl = "$webImagePath/$imageName"
            
            $linkMap[$imageName] = $imageUrl
            
            $imageNameWithoutExt = [System.IO.Path]::GetFileNameWithoutExtension($imageName)
            if ($imageNameWithoutExt -ne $imageName) {
                $linkMap[$imageNameWithoutExt] = $imageUrl
            }
        }
    }
    
    Write-Host "이미지 파일 매핑 완료" -ForegroundColor Green
} else {
    Write-Host "이미지 폴더를 찾을 수 없습니다: $imagesFullPath" -ForegroundColor Yellow
}

Write-Host "생성된 URL 매핑 개수: $($linkMap.Count)" -ForegroundColor Green
Write-Host ""

# =================================================================
# 3단계: 프론트 매터 완전 재작성 및 위키링크, 이미지 경로 변환
# =================================================================
Write-Host "=== 3단계: 프론트 매터 완전 재작성 및 위키링크, 이미지 경로 변환 시작 ===" -ForegroundColor Magenta

$totalConvertedLinks = 0
$totalConvertedImages = 0
$totalFilesProcessed = 0
$problematicItems = @()

foreach ($section in $sections) {
    $sectionPath = Join-Path $contentPath $section
    
    if (Test-Path $sectionPath) {
        Write-Host "섹션 처리 중: $section" -ForegroundColor Green
        
        $allFolders = @($sectionPath) + (Get-ChildItem -Path $sectionPath -Directory -Recurse | Select-Object -ExpandProperty FullName)
        
        foreach ($folderPath in $allFolders) {
            $mdFiles = Get-ChildItem -Path $folderPath -Filter "*.md" -File | Sort-Object Name
            
            if ($mdFiles.Count -gt 0) {
                $relativePath = $folderPath.Replace($contentPath, "").TrimStart('\')
                Write-Host "  폴더: /$relativePath ($($mdFiles.Count)개 파일)" -ForegroundColor Cyan
                
                $weight = 1
                foreach ($file in $mdFiles) {
                    Write-Host "    - 처리 중: $($file.Name)"
                    $totalFilesProcessed++
                    
                    $content = Get-Content -Path $file.FullName -Raw -Encoding UTF8
                    $fileCreationTime = (Get-Item $file.FullName).CreationTime
                    $fileModifiedTime = (Get-Item $file.FullName).LastWriteTime
                    
                    # ===================================================
                    # 3.1: 기존 프론트 매터 및 메타 태그 완전 제거
                    # ===================================================
                    $cleanedBody = $content
                    
                    if ($content -match '(?s)^---\s*\n.*?\n---\s*\n(.*)$') {
                        $cleanedBody = $Matches[1]
                    }
                    
                    $cleanedBody = $cleanedBody -replace '(?s)<div\s+style="display:none">\s*<meta\s+property="article:published_time"[^>]*>\s*<meta\s+property="article:modified_time"[^>]*>\s*</div>[\r\n]*', ''
                    $cleanedBody = $cleanedBody -replace '<meta property="article:(published|modified)_time"[^>]*>[\r\n]*', ''
                    $cleanedBody = $cleanedBody.Trim()

                    # Title 결정
                    $newTitle = ""
                    if ($file.Name -eq "_index.md") {
                        $newTitle = Split-Path $file.DirectoryName -Leaf
                    } else {
                        $newTitle = [System.IO.Path]::GetFileNameWithoutExtension($file.Name)
                    }
                    
                    $finalCreatedDate = $fileCreationTime.ToString("yyyy-MM-ddTHH:mm:sszzz")
                    $finalModifiedDate = $fileModifiedTime.ToString("yyyy-MM-ddTHH:mm:sszzz")
                    
                    # ===================================================
                    # 3.2: 새로운 프론트 매터 구성 (sidebar: open: false 반영)
                    # ===================================================
                    $newFMContent = @()
                    $newFMContent += "title: `"$newTitle`""
                    $newFMContent += "date: $finalCreatedDate"
                    $newFMContent += "lastmod: $finalModifiedDate"
                    $newFMContent += "type: docs"
                    $newFMContent += "sidebar:"
                    $newFMContent += "  open: false" # <--- 여기를 false로 변경
                    $newFMContent += "weight: $weight"
                    
                    # ===================================================
                    # 3.3: 위키링크 변환 ([[target|alias]] 또는 [[image.png]])
                    # ===================================================
                    $fileConvertedWikilinks = 0
                    $wikiLinkPattern = '\[\[([^\|\]]+)(?:\|([^\]]+))?\]\]'
                    
                    try {
                        $matches = [regex]::Matches($cleanedBody, $wikiLinkPattern)
                        
                        if ($matches.Count -gt 0) {
                            foreach ($match in $matches | Sort-Object -Descending { $_.Index }) {
                                $fullMatch = $match.Value
                                $target = $match.Groups[1].Value.Trim()
                                $alias = if ($match.Groups.Count -gt 2) { $match.Groups[2].Value.Trim() } else { "" }
                                
                                $linkText = if ($alias) { $alias } else { $target }
                                
                                $isImageWikilink = $target -match '\.(png|jpg|jpeg|gif|webp|svg|bmp)$'
                                
                                if ($target -match '%[^0-9A-Fa-f]|%$|%[0-9A-Fa-f]$') {
                                    $problematicItems += @{
                                        File = $file.FullName
                                        Item = $fullMatch
                                        Type = "Wikilink (Problematic Chars)"
                                        Target = $target
                                    }
                                    $cleanedBody = $cleanedBody.Remove($match.Index, $match.Length).Insert($match.Index, $linkText)
                                    continue
                                }
                                
                                $normalizedTarget = $target -replace '[-_]', ' '
                                $targetUrl = $null
                                
                                if ($linkMap.ContainsKey($target)) {
                                    $targetUrl = $linkMap[$target]
                                } elseif ($linkMap.ContainsKey($normalizedTarget)) {
                                    $targetUrl = $linkMap[$normalizedTarget]
                                } elseif ($target.Contains('/')) {
                                    $baseTargetName = Split-Path $target -Leaf
                                    if ($linkMap.ContainsKey($baseTargetName)) {
                                        $targetUrl = $linkMap[$baseTargetName]
                                    } elseif ($linkMap.ContainsKey($baseTargetName -replace '[-_]', ' ')) {
                                        $targetUrl = $linkMap[$baseTargetName -replace '[-_]', ' ']
                                    }
                                }

                                if ($targetUrl) {
                                    if ($isImageWikilink) {
                                        $altText = if ($alias) { $alias } else { [System.IO.Path]::GetFileNameWithoutExtension($target) }
                                        $altText = $altText -replace '\s+', '-'
                                        $altText = $altText.Trim('-')
                                        $markdownLink = "![$altText]($targetUrl)"
                                        $fileConvertedWikilinks++
                                        $totalConvertedImages++
                                    } else {
                                        $markdownLink = "[$linkText]($targetUrl)"
                                        $fileConvertedWikilinks++
                                        $totalConvertedLinks++
                                    }
                                    $cleanedBody = $cleanedBody.Remove($match.Index, $match.Length).Insert($match.Index, $markdownLink)
                                } else {
                                    $problematicItems += @{
                                        File = $file.FullName
                                        Item = $fullMatch
                                        Type = "Wikilink (Unmapped)"
                                        Target = $target
                                    }
                                    $cleanedBody = $cleanedBody.Remove($match.Index, $match.Length).Insert($match.Index, $linkText)
                                }
                            }
                        }
                    }
                    catch {
                        Write-Host "      ⚠ 위키링크 처리 중 오류 발생: $($_.Exception.Message)" -ForegroundColor Red
                    }

                    # ===================================================
                    # 3.4: ![](이미지파일명.png) 형태의 이미지 경로 변환
                    # ===================================================
                    $fileConvertedSimpleImages = 0
                    
                    try {
                        $imageExtensions = @("png", "jpg", "jpeg", "gif", "webp", "svg", "bmp")
                        
                        foreach ($ext in $imageExtensions) {
                            $pattern = "![]("
                            
                            $startIndex = 0
                            while ($true) {
                                $index = $cleanedBody.IndexOf($pattern, $startIndex)
                                if ($index -eq -1) { break }
                                
                                $endIndex = $cleanedBody.IndexOf(")", $index + $pattern.Length)
                                if ($endIndex -eq -1) { break }
                                
                                $filename = $cleanedBody.Substring($index + $pattern.Length, $endIndex - $index - $pattern.Length)
                                
                                if ($filename -match "\.($($imageExtensions -join '|'))$" -and -not $filename.StartsWith("/")) {
                                    $oldImageTag = "$pattern$filename)"
                                    $newImageTag = "![]($webImagePath/$filename)"
                                    
                                    $cleanedBody = $cleanedBody.Replace($oldImageTag, $newImageTag)
                                    $fileConvertedSimpleImages++
                                    $totalConvertedImages++
                                    
                                    $startIndex = $index + $newImageTag.Length
                                } else {
                                    $startIndex = $endIndex + 1
                                }
                            }
                        }
                    }
                    catch {
                        Write-Host "      ⚠ 이미지 처리 중 오류 발생: $($_.Exception.Message)" -ForegroundColor Red
                    }

                    if ($fileConvertedWikilinks -gt 0) {
                        Write-Host "      ✓ $($fileConvertedWikilinks)개의 위키링크 변환 완료." -ForegroundColor Green
                    }
                    if ($fileConvertedSimpleImages -gt 0) {
                        Write-Host "      ✓ $($fileConvertedSimpleImages)개의 '![](파일명)' 변환 완료." -ForegroundColor Green
                    }

                    # ===================================================
                    # 3.5: SEO 메타 태그 생성
                    # ===================================================
                    try {
                        $publishedISO = ([DateTime]::Parse($finalCreatedDate)).ToUniversalTime().ToString("yyyy-MM-ddTHH:mm:ssZ")
                        $modifiedISO = ([DateTime]::Parse($finalModifiedDate)).ToUniversalTime().ToString("yyyy-MM-ddTHH:mm:ssZ")
                    } catch {
                        $publishedISO = $fileCreationTime.ToUniversalTime().ToString("yyyy-MM-ddTHH:mm:ssZ")
                        $modifiedISO = $fileModifiedTime.ToUniversalTime().ToString("yyyy-MM-ddTHH:mm:ssZ")
                    }
                    
                    $metaTagsHtml = @"

<div style="display:none">
  <meta property="article:published_time" content="$publishedISO" />
  <meta property="article:modified_time" content="$modifiedISO" />
</div>
"@
                    
                    # ===================================================
                    # 3.6: 파일 저장
                    # ===================================================
                    $finalFrontMatterBlock = "---`n" + ($newFMContent -join "`n") + "`n---"
                    $finalContent = $finalFrontMatterBlock + $metaTagsHtml + "`n" + $cleanedBody
                    
                    Set-Content -Path $file.FullName -Value $finalContent -Encoding UTF8
                    
                    Write-Host "        ✓ (Weight: $weight) 완료" -ForegroundColor Green
                    
                    $weight++
                }
            }
        }
    } else {
        Write-Host "경로를 찾을 수 없습니다: $sectionPath - 스킵합니다." -ForegroundColor Yellow
    }
}

Write-Host ""
Write-Host "========================================" -ForegroundColor Magenta
Write-Host "   모든 자동화 작업 완료!" -ForegroundColor Magenta
Write-Host "   ✓ 누락된 _index.md 파일 자동 생성" -ForegroundColor Magenta
Write-Host "   ✓ 기존 프론트 매터 완전 삭제 및 재작성" -ForegroundColor Magenta
Write-Host "   ✓ Permalinks 기반 URL 매핑 적용" -ForegroundColor Magenta
Write-Host "   ✓ 위키링크 ([[]]) 변환 지원" -ForegroundColor Magenta
Write-Host "   ✓ '![](파일명)' 형태 이미지 경로 변환 지원" -ForegroundColor Magenta
Write-Host "   ✓ 총 $($totalConvertedLinks)개의 일반 위키링크 변환" -ForegroundColor Magenta
Write-Host "   ✓ 총 $($totalConvertedImages)개의 이미지 링크 변환" -ForegroundColor Magenta
Write-Host "   ✓ 총 $($totalFilesProcessed)개의 파일 처리" -ForegroundColor Magenta

if ($problematicItems.Count -gt 0) {
    Write-Host "" -ForegroundColor Yellow
    Write-Host "⚠ 문제성 항목 (매핑 실패 또는 잘못된 형식) 목록:" -ForegroundColor Yellow
    foreach ($item in $problematicItems) {
        Write-Host "  - 파일: $($item.File)" -ForegroundColor Yellow
        Write-Host "    유형: $($item.Type)" -ForegroundColor Yellow
        Write-Host "    항목: $($item.Item)" -ForegroundColor Yellow
        if ($item.Target) { Write-Host "    타겟: $($item.Target)" -ForegroundColor Yellow }
    }
}

Write-Host "   이제 다음 단계를 진행하세요:" -ForegroundColor Magenta
Write-Host "   1. git add ." -ForegroundColor White
